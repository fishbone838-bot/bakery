async function load() { try { // 讀取設定 const config = await getStockConfig(); document.getElementById("maxInput").value = config.max; // 讀取所有訂單 const orders = await getOrders(); const list = document.getElementById("list"); list.innerHTML = ""; const today = getTodayStr(); let todaySold = 0; orders.forEach(o => { // 1. 計算今天的銷量 if (o.date === today) { todaySold += Number(o.amount); } // 2. 顯示所有訂單 (舊訂單也會顯示在這裡) const tr = document.createElement("tr"); tr.innerHTML = ` ${o.number} ${o.name} ${o.phone} ${o.amount} ${o.time?.toDate().toLocaleString() || (o.date || "")} `; list.appendChild(tr); }); document.getElementById("adminStockHint").innerText = `今日統計：已收單 ${todaySold} 份 / 總上限 ${config.max} 份`; } catch (err) { console.error("載入失敗:", err); } } async function saveMax() { const newMax = document.getElementById("maxInput").value; await updateStockConfig(newMax); // 這次 main.js 有定義了，不會再報錯 alert("儲存成功！"); load(); } load();
